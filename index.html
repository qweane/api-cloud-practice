const button = document.getElementById('loadBtn'); const list = document.getElementById('userList'); button.addEventListener('click', async () => { button.innerText = 'Загрузка...'; // Запрос данных с PokeAPI const response = await fetch('https://pokeapi.co/api/v2/pokemon/pikachu'); const pokemon = await response.json(); list.innerHTML = ''; // Создаем элемент списка и выводим имя и базовый вес сущности const li = document.createElement('li'); li.innerText = pokemon.name + ' (Weight: ' + pokemon.weight + ')'; list.appendChild(li); button.innerText = 'Данные загружены!'; });
